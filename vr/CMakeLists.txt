cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(netdisplay_vr LANGUAGES C CXX)
  include(CTest)
endif()
enable_language(CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(../cmake/NetDisplayCore.cmake)
nd_add_video_library()
find_package(Vulkan REQUIRED)
pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)
set(OPENVR_INCLUDE_DIR "" CACHE PATH "OpenVR SDK headers")
if(NOT EXISTS "${OPENVR_INCLUDE_DIR}/openvr_driver.h")
  message(FATAL_ERROR "Set OPENVR_INCLUDE_DIR to the OpenVR SDK headers")
endif()
set(PACKAGE "${CMAKE_BINARY_DIR}/netdisplay-vr")
file(GLOB_RECURSE CAPTURE_SOURCES CONFIGURE_DEPENDS vendor/capture/*.cpp)
add_library(netdisplay_capture SHARED ${CAPTURE_SOURCES})
target_include_directories(netdisplay_capture PRIVATE vendor/capture)
target_include_directories(netdisplay_capture SYSTEM PRIVATE "${Vulkan_INCLUDE_DIR}")
target_compile_definitions(netdisplay_capture PRIVATE VK_USE_PLATFORM_XLIB_XRANDR_EXT)
target_link_libraries(netdisplay_capture PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(netdisplay_capture PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${PACKAGE}/capture")
configure_file(vendor/capture/layer/alvr_x86_64.json "${PACKAGE}/capture/netdisplay.json" COPYONLY)
add_library(netdisplay_drm_shim SHARED vendor/drm-lease-shim.cpp)
target_link_libraries(netdisplay_drm_shim PRIVATE PkgConfig::DRM ${CMAKE_DL_LIBS})
set_target_properties(netdisplay_drm_shim PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${PACKAGE}/capture")
add_library(driver_netdisplay_vr SHARED ../steamvr/driver.cpp)
target_include_directories(driver_netdisplay_vr SYSTEM PRIVATE "${OPENVR_INCLUDE_DIR}")
target_compile_definitions(driver_netdisplay_vr PRIVATE NETDISPLAY_VR_DIRECT=1)
target_link_libraries(driver_netdisplay_vr PRIVATE PkgConfig::SODIUM Threads::Threads)
set_target_properties(driver_netdisplay_vr PROPERTIES PREFIX "" BUILD_RPATH "$ORIGIN"
  LIBRARY_OUTPUT_DIRECTORY "${PACKAGE}/driver/bin/linux64")
add_custom_command(TARGET driver_netdisplay_vr POST_BUILD
  COMMAND "${CMAKE_COMMAND}" "-DDRIVER=$<TARGET_FILE:driver_netdisplay_vr>"
    "-DSODIUM_DIRS=${SODIUM_LIBRARY_DIRS}" -P "${CMAKE_CURRENT_SOURCE_DIR}/../steamvr/bundle-sodium.cmake")
configure_file(driver.vrdrivermanifest "${PACKAGE}/driver/driver.vrdrivermanifest" COPYONLY)
add_executable(netdisplay-vr-server server.c capture.cpp)
target_include_directories(netdisplay-vr-server PRIVATE ../src vendor/capture)
target_compile_options(netdisplay-vr-server PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
target_link_libraries(netdisplay-vr-server PRIVATE nd_video Vulkan::Vulkan)
set_target_properties(netdisplay-vr-server PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PACKAGE}/bin")
configure_file(netdisplay-vr.py "${PACKAGE}/netdisplay-vr" COPYONLY)
configure_file(compositor-wrapper.sh "${PACKAGE}/compositor-wrapper" COPYONLY)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME vr-launcher COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/launcher_test.py")
  add_test(NAME vr-control COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/control_test.py" $<TARGET_FILE:netdisplay-vr-server>)
  include(../cmake/NetDisplayMotionTests.cmake)
  if(NOT TARGET netdisplay-video-tx-test)
    include(../cmake/NetDisplayVideoTests.cmake)
  endif()
endif()

if(EXISTS "${OPENVR_INCLUDE_DIR}/openvr.h")
  add_executable(netdisplay-vr-diagnose diagnose.cpp)
  target_include_directories(netdisplay-vr-diagnose SYSTEM PRIVATE "${OPENVR_INCLUDE_DIR}")
  target_link_libraries(netdisplay-vr-diagnose PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
  set_target_properties(netdisplay-vr-diagnose PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PACKAGE}/bin")
endif()
